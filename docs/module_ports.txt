// Widths: cube [71:0] = 24 stickers x 3 bits, stage [2:0], move [4:0] (codes 0-17),
// colour [2:0], sticker index [4:0] (face*4 + tile), view [6:0],
// vga_rgb [11:0] = {r[3:0], g[3:0], b[3:0]} (same layout as colour_to_rgb())
// Button order everywhere: {C, U, L, R, D}
// Ticks are named by rate: tick_25mhz, tick_1khz, tick_4hz are 1-cycle enable
// pulses on the 100 MHz clock. clk_6p25mhz is a real clock, OLED driver only.

module top_basys3(
    input clk, // 100 MHz board clock
    input btnC, btnU, btnL, btnR, btnD, // buttons
    input [15:0] sw, // slide switches
    inout PS2Clk, PS2Data, // mouse clock, data
    output [15:0] led, // LEDs
    output [6:0] seg, // 7-seg segments
    output dp, // 7-seg decimal point
    output [3:0] an, // 7-seg anodes
    output [3:0] vgaRed, vgaGreen, vgaBlue, // VGA pins (XDC names), {vgaRed, vgaGreen, vgaBlue} = vga_rgb
    output Hsync, Vsync, // VGA syncs
    output [7:0] JC // Pmod OLED pins
);

// cube_defs.vh (src/pkg) is a header, not a module: no ports.
// Each module does `include "cube_defs.vh" inside the module, after its port list.
// Holds stage names, move codes, colour codes, tick dividers, SOLVED_CUBE, get_tile(), set_tile(), colour_to_rgb()

module tick_gen(
    input clk, // 100 MHz clock
    input rst, // reset
    input en_25mhz, en_1khz, en_4hz, // enable each tick
    output tick_25mhz, // 1-cycle pulse every 4 clks (VGA pixel)
    output tick_1khz, // 1-cycle pulse every 100,000 clks (7-seg multiplex)
    output tick_4hz, // 1-cycle pulse every 25,000,000 clks (scramble/solve playback, LED blink)
    output clk_6p25mhz // real 6.25 MHz clock, only for the provided OLED driver
);

module game_fsm(

    input rst, // reset

    input manual_valid, scr_valid, sol_valid, // matching ready pulses

module cube_engine(

    input rst, // reset
    input cube_reset, // reset cube to solved
    input [4:0] move_req, // move to apply
    input move_valid, // apply move this cycle
    output [71:0] cube, // live cube state
    output [15:0] move_count // moves applied since reset
);

module move_rules(
    input [71:0] cube_in, // cube before move
    input [4:0] move, // move to apply
    output [71:0] cube_out // cube after move

module scrambler(
    input clk, // 100 MHz clock
    input rst, // reset
    input start, // pulse to begin scramble
    input tick_4hz, // one move per tick
    output [4:0] scr_move, // next scramble move
    output scr_valid, // pulse when scr_move is ready
    output busy, // scramble in progress
    output done // pulse when 25 moves are sent
);

module solver(
    input clk, // 100 MHz clock
    input rst, // reset
    input start, cancel, // pulses: begin solving, abort
    input tick_4hz, // one move per tick during playback
    input [71:0] cube, // cube to solve
    output [4:0] sol_move, // next solution move
    output sol_valid, // pulse when sol_move is ready
    output busy, // searching or playing back
    output done, err, // pulses: cube solved, timeout or cancel
    output [3:0] moves_left // solution moves remaining
);

module camera_unit(
    input clk, // 100 MHz clock
    input rst, // reset
    input [2:0] stage, // current game stage
    input [4:0] btn_pressed, // button pulses
    input mouse_l, // left button held (drag)
    input signed [8:0] mouse_dx, mouse_dy, // movement this packet
    input mouse_event, // new mouse packet
    output [7:0] camera_angle_id // {pitch_bin[3:0], yaw_bin[3:0]}, 0-255
);

Camera-angle IDs form a 16-by-16 grid. Bits [3:0] are the yaw bin and bits
[7:4] are the pitch bin. Each bin ranges from 0 to 15.

module decode_mem(
    input clk, // 100 MHz clock
    input [7:0] camera_angle_id, // {pitch_bin[3:0], yaw_bin[3:0]}
    input [3:0] visible_sticker_slot, // requested depth-ordered slot, 0-11
    output [4:0] actual_sticker_id, // live cube sticker index, or 31 for empty
    output [79:0] sticker_corner_coords, // four {x[9:0], y[9:0]} corners
    output [107:0] sticker_edge_slopes // four signed 27-bit Q16 slopes
);

The angle decoder table has 256 entries. Each 10-bit entry is
{projection_id[4:0], cube_rotation_id[4:0]}; projection IDs are 0-27 and
rotation IDs are 0-23. Each projection has 12 visible sticker slots ordered
front to back, so slot 0 is the frontmost. The table stores reference sticker
IDs; decode_mem remaps them to the actual sticker indices used by cube.
ID 31 marks an empty slot.

Corners are packed from least to most significant as {x0,y0}, {x1,y1},
{x2,y2}, {x3,y3}, with 10 bits per coordinate. Slopes occupy four signed
27-bit Q16 fields in edge order 0-to-1, 1-to-2, 2-to-3, and 3-to-0.

The synchronous lookup response corresponds to the request sampled three
rising clock edges earlier. The slot and camera-angle request must stay
aligned with that response.

module cube_renderer(
    input clk, // 100 MHz clock
    input rst, // reset all renderer state
    input tick_25mhz, // one-cycle pulse at the VGA pixel rate
    input frame_start, // pulse at the start of a new frame
    input [7:0] camera_angle_id, // {pitch_bin[3:0], yaw_bin[3:0]}
    input [9:0] pix_x, pix_y, // current VGA pixel position
    input [71:0] cube, // live sticker colours, 24 entries x 3 bits
    output [2:0] pixel_colour, // colour code of this pixel
    output pixel_on_cube, // 1 when this pixel is on a sticker
    output row_done // one-cycle pulse when a complete row is ready
);

Sticker index s is face*4 + tile and addresses cube[s*3 +: 3]. On
frame_start, the renderer latches camera_angle_id and requests all 12 slots
from decode_mem. It stores each returned actual_sticker_id,
sticker_corner_coords, and sticker_edge_slopes in the matching front-to-back
slot. Geometry is ready after all 12 responses arrive. row_done pulses for
one clock when a completed scanline has been stored and its row buffer is
valid.
module calc_sticker_span(
    input clk,
    input rst,
    input start, // start one sticker calculation
    input [9:0] row_y, // screen row to check
    input [79:0] corners, // {x3,y3,...,x0,y0}, 10 bits per coordinate
    input [107:0] slopes, // four signed Q16 x-per-row slopes
    input [107:0] edge_state_in, // four signed Q16 edge x positions
    output [107:0] edge_state_out, // edge positions advanced for the next row
    output [9:0] span_left, span_right, // rounded pixel bounds; 640/0 means invalid
    output span_valid, // one when at least one pixel crosses the sticker
    output busy,
    output done // one-cycle pulse when results are ready
);

The helper checks four edges sequentially. It includes each edge's lower-y
endpoint and excludes its higher-y endpoint, rounds the left bound up and the
right bound down, and updates saved edge positions for the next row.
module vga_unit(
    input clk, // 100 MHz clock
    input rst, // reset
    input tick_25mhz, // pixel pulse
    input [2:0] stage, // current game stage
    input [2:0] pixel_colour, // cube colour from renderer
    input pixel_on_cube, // renderer hit flag
    input [9:0] mouse_x, mouse_y, // cursor position
    output [9:0] pix_x, pix_y, // current pixel position
    output frame_start, // pulse at start of vblank
    output [11:0] vga_rgb, // {r[3:0], g[3:0], b[3:0]}
    output hsync, vsync // VGA syncs
);
